### 0.1 三类表，写入规则不同

| 类 | 规则 | 怎么强制 |

```sql
        TG_TABLE_NAME, TG_OP;
```

### 2.1 计划与格子

```sql
CREATE TABLE scm.plan (
    plan_id      bigserial PRIMARY KEY,
    title        text        NOT NULL,
    period_start date        NOT NULL,          -- 必须月初
    months       int         NOT NULL CHECK (months BETWEEN 1 AND 12),
    owner_actor  text        NOT NULL REFERENCES scm.actor(actor_id),
    archived_at  timestamptz,
    created_by   text        NOT NULL,
    created_at   timestamptz NOT NULL DEFAULT now(),
    CHECK (date_trunc('month', period_start) = period_start)
);

-- ★ 草稿格子：可改、无状态。生效值 = COALESCE(planned_units, system_units)
CREATE TABLE scm.plan_cell (
    plan_id       bigint NOT NULL REFERENCES scm.plan(plan_id) ON DELETE CASCADE,
    seller_id     text   NOT NULL REFERENCES scm.seller(seller_id),
    sku           text   NOT NULL REFERENCES scm.sku_catalog(sku),
    period_start  date   NOT NULL,
    system_units  int    CHECK (system_units  >= 0),   -- 预测函数算出，向上取整
    planned_units int    CHECK (planned_units >= 0),   -- 运营填的「计划销量」
    updated_by    text,
    updated_at    timestamptz NOT NULL DEFAULT now(),
    PRIMARY KEY (plan_id, seller_id, sku, period_start)
);

-- ★ A-6：格子由哪几个 msku 撑着（只算已认领的）
CREATE TABLE scm.plan_cell_msku (
    plan_id      bigint NOT NULL,
    seller_id    text   NOT NULL,
    sku          text   NOT NULL,
    period_start date   NOT NULL,
    seller_sku   text   NOT NULL,
    sid          text   NOT NULL,
    system_units int    NOT NULL CHECK (system_units >= 0),
    PRIMARY KEY (plan_id, seller_id, sku, period_start, seller_sku, sid),
    FOREIGN KEY (plan_id, seller_id, sku, period_start)
        REFERENCES scm.plan_cell ON DELETE CASCADE
);
```

★ **`plan_cell_msku` 存的是构成，不是另一个真相。**
`plan_cell.system_units` 必须等于其构成之和 —— 由 §6 的约束触发器守。
没有它，界面上「这 300 件由哪几个 listing 撑着」答不出来，
**而数对不上时无从解释**。

## 3. ★ 状态机四件套（`04` §7.1 / §7.6b 的兑现）

四个状态机**共用同一个模式**。这里给 S1 全文，其余三个只列白名单内容。

```sql
-- ① 白名单：状态机本身是数据
CREATE TABLE scm.plan_line_transition (
    from_state   text    NOT NULL,
    to_state     text    NOT NULL,
    kind         text    NOT NULL CHECK (kind IN ('forward','back','cancel')),
    needs_reason boolean NOT NULL DEFAULT false,
    PRIMARY KEY (from_state, to_state)
);

INSERT INTO scm.plan_line_transition VALUES
 ('已提交','已确认','forward',false),
 ('已确认','已下单','forward',false),
 ('已下单','准备排货','forward',false),
 ('准备排货','已排货','forward',false),
 ('已排货','已完结','forward',false),
 ('已确认','已提交','back',   false),          -- ★ 全服务唯一一条退回
 ('已提交','已撤销','cancel', true),
 ('已确认','已撤销','cancel', true),
 ('已下单','已撤销','cancel', true),
 ('准备排货','已撤销','cancel', true),
 ('已排货','已撤销','cancel', true);
-- ★ 表里没有以 已完结/已撤销 作 from_state 的行 → 终态不可离开，由外键裁决

-- ② 事件表：唯一能改状态的入口
CREATE TABLE scm.plan_line_event (
    seq        bigserial PRIMARY KEY,           -- ★ 序列，不用时间戳
    line_id    bigint NOT NULL REFERENCES scm.plan_line(line_id),
    from_state text   NOT NULL,
    to_state   text   NOT NULL,
    actor      text   NOT NULL,
    reason     text,
    src        text,                            -- 触发来源：哪张采购单/排货单/哪轮回执
    at         timestamptz NOT NULL DEFAULT now(),
    FOREIGN KEY (from_state, to_state)
        REFERENCES scm.plan_line_transition     -- ★ 非法迁移 = 外键违反
);
CREATE TRIGGER t_append_only BEFORE UPDATE OR DELETE ON scm.plan_line_event
    FOR EACH ROW EXECUTE FUNCTION scm.forbid_update_delete();
```

### 3.1 其余三个白名单的内容

| 状态机 | 白名单行 |
|---|---|
| **`consignment_transition`**（我方进度） | `草稿→已确认`(f) · `已确认→已投送`(f) · `已确认→草稿`(**back**) · `已投送→已归档`(f) · `草稿→已放弃`(cancel) |
| `sheet_transition` | `草稿→已确认`(f) · `草稿→已取消`(cancel) |
| `po_transition` | `草稿→已下单`(f) |

★ **`consignment_transition` 里没有 `已投送→草稿`，也没有 `已投送→已放弃`。**
跨过 `SendInbound` / `SendGoods` 之后货已在路上、库存已扣 ——
**画得出而现实中退不回，比不画更糟：代码会去调那条不存在的路径。**

### 3.2 三个通用触发器

```sql
-- ⓐ 乐观校验 + 写物化列（同时解决并发）
CREATE FUNCTION scm.sync_plan_line_state() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE cur text;
BEGIN
    SELECT state INTO cur FROM scm.plan_line WHERE line_id = NEW.line_id FOR UPDATE;
    IF cur IS DISTINCT FROM NEW.from_state THEN
        RAISE EXCEPTION 'from_state 不匹配：当前 %，事件称 %', cur, NEW.from_state;
    END IF;
    PERFORM set_config('app.in_state_sync','on',true);
    UPDATE scm.plan_line SET state = NEW.to_state WHERE line_id = NEW.line_id;
    PERFORM set_config('app.in_state_sync','off',true);
    RETURN NEW;
END $$;

-- ⓑ 理由必填
CREATE FUNCTION scm.require_reason() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE need boolean;
BEGIN
    SELECT needs_reason INTO need FROM scm.plan_line_transition
     WHERE from_state = NEW.from_state AND to_state = NEW.to_state;
    IF need AND coalesce(btrim(NEW.reason),'') = '' THEN
        RAISE EXCEPTION '这条迁移必须写理由：% → %', NEW.from_state, NEW.to_state;
    END IF;
    RETURN NEW;
END $$;

-- ⓒ 禁止直改 state
CREATE FUNCTION scm.state_must_go_through_event() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
    IF current_setting('app.in_state_sync', true) IS DISTINCT FROM 'on' THEN
        RAISE EXCEPTION '状态只能由事件表推进，不许直接 UPDATE state';
    END IF;
    RETURN NEW;
END $$;
```

### 4.1 ★ 承重墙① —— 采购来源映射

> 「多少件是为谁准备的」的记账表。见 `02` §〇。

```sql
CREATE TABLE scm.po_source_map (
    po_id     bigint NOT NULL,
    line_no   int    NOT NULL,
    line_id   bigint NOT NULL REFERENCES scm.plan_line(line_id),
    -- ★★ 店铺归属落在**这里**，不在 plan_line 上（P2 保持不变）。
    --    它记的是「这次组单消耗的是哪个店那份需求」——
    --    是**采购员勾出来的决定**，不是系统按比例摊的，所以它是事实。
    seller_id text   NOT NULL REFERENCES scm.seller(seller_id),
    units     int    NOT NULL CHECK (units > 0),    -- ★ 关系上挂着的那个数
    PRIMARY KEY (po_id, line_no, line_id, seller_id),
    FOREIGN KEY (po_id, line_no) REFERENCES scm.po_line ON DELETE CASCADE
);

-- ★ A-3：两个方向都建索引
CREATE INDEX po_source_by_line_idx   ON scm.po_source_map (line_id);              -- 求和方向
CREATE INDEX po_source_by_seller_idx ON scm.po_source_map (line_id, seller_id);   -- ★ 分店看
-- （主键前缀已覆盖 po_id,line_no 方向 —— 下钻方向）
```

#### ★ 裁定：「看的时候分店，下单的时候合并」（负责人 2026-09-18）

```
【候选】按 计划记录 × 店铺 展开，可分别勾
  PLAN-Q4  DCC1800035  US 2026-10  需求 40  未组单 15  [选]
  PLAN-Q4  DCC1800035  DE 2026-10  需求 20  未组单 20  [选]
                    ↓ 勾选后
【采购单】按货号合并成一行下给供应商
  DCC1800035  2026-10  数量 35  ← 底下是 2 条 po_source_map 记录
```

★ **为什么这不违反「不自动分摊」（裁定 C4）**：

| | 库存预测里的「未定向量」 | 这里的店铺归属 |
|---|---|---|
| 怎么来的 | ★ 系统**按规则假设**出来的 | ★ **人勾出来的** |
| 性质 | 假设（必须标明规则、可改） | **事实**（是一次决定的留痕） |

**新增恒等式**：

```
Σ_店铺 未组单量[line, 店铺] ≡ 货号级未组单量[line]
其中 未组单量[line,s] = plan_line.demand_by_seller[s]
                      − Σ po_source_map(line, s).units
```

★ 这条自然成立（两边都是同一批行的求和），**但仍要有断言** ——
它成立的前提是「每一次组单都写了 seller_id」，而 `NOT NULL` 只挡住了写空，
挡不住**写错店**。核对方式见 §7.3 的同类做法。

⚠️ **上限约束**：`Σ_s po_source_map(line,s).units ≤ plan_line.demand_by_seller[s]` 是
**跨行聚合**，普通 CHECK 做不到 —— 要走 §6.1 的约束触发器，和承重墙的木桶判据同一套。

#### ★★ 承重墙③ 排货行 ↔ 货件

```
Σ 货件行.units (sku, seller)  ≡  排货行.units (sku, seller)
```

★ 与 ①② 的根本不同：**这一面的数据不是我们写的，是回读来的。** 由此：

| # | 约束 | 违反的后果 |
|---|---|---|
| 1 | ★ 恒等式按 `units`（= `quantity_shipped`）立 | 拿到货量当投出量，会把在途算成已到 |
| 2 | ★ `units_received` 用**下限判据**（外部物理事实） | 到货一件就当全到 |
| 3 | ★ 允许「已投送 · 归属未知」—— 并且**必须点名**，不许静默当 0 | 「还没采到」会被读成「没拆」 |
| 4 | ⚠️ `from_wid` 对不回去时**不许猜分摊**（C-9） | 平摊出来的归属没人能证伪 |

**统一形状的视图** —— 单份时投影，多份时读明细：

```sql
CREATE VIEW scm.v_split_attribution AS
WITH n AS (SELECT consignment_id, count(*) AS splits
           FROM scm.consignment_split GROUP BY 1)
-- 单份：整条排货行归这一份（海外仓恒走这支）
SELECT sp.split_id, l.consignment_id, l.sku, l.seller_id, l.from_wid, l.units,
       'projected'::text AS basis
  FROM scm.consignment_line l
  JOIN n ON n.consignment_id = l.consignment_id AND n.splits = 1
  JOIN scm.consignment_split sp ON sp.consignment_id = l.consignment_id
UNION ALL
-- 多份：只认回读来的明细
SELECT sl.split_id, sp.consignment_id, sl.sku, sl.seller_id, NULL::int, sl.units,
       'observed'::text
  FROM scm.consignment_split_line sl
  JOIN scm.consignment_split sp ON sp.split_id = sl.split_id
  JOIN n ON n.consignment_id = sp.consignment_id AND n.splits > 1;
```

★ `basis` 列是刻意的：**一个数要能回答「它是关于什么的」** ——
`projected` 是我们推的，`observed` 是回读的事实，二者不许长得一样。

**核对视图** —— 差额与「归属未知」都要点名：

```sql
CREATE VIEW scm.v_shipment_check AS
SELECT l.consignment_id, l.sku, l.seller_id,
       l.units                              AS line_units,
       coalesce(sum(a.units), 0)            AS attributed_units,
       l.units - coalesce(sum(a.units), 0)  AS gap,
       count(a.split_id)                    AS splits_seen,
       min(a.basis)                         AS basis
  FROM scm.consignment_line l
  LEFT JOIN scm.v_split_attribution a
         ON a.consignment_id = l.consignment_id
        AND a.sku = l.sku AND a.seller_id = l.seller_id
 GROUP BY l.consignment_id, l.sku, l.seller_id, l.units;
```

★ `gap != 0` 与 `splits_seen = 0` 是**两种不同的病**，视图必须分得开：
前者是拆分对不上，后者是**还没回读到** —— 处置相反。

### 5.0 ★★ 一个步骤的产出，在后续每一步里都必须 NOT NULL

> **这一节是被一个真 bug 逼出来的**（负责人 2026-09-19 实测）：
> 排货表头写着「已编排 3 行 / 52 件」，而排货网格上**每一格都是「已安排 0 件」**。
> 货在表里，网格上一件都看不见。

**根因**：`consignment_line` 上**没有店铺归属**（本文档此前就漏了这一列），
于是「这批货给哪个店」这个结论在编入排货单时被丢掉，
而网格是按 `货号 × 店铺` 组织的 —— 归属丢了，这笔货就落不到任何一格。

★ **「已安排 0 件」和「这批货没被算进来」长得一模一样。**

#### 判据

```
排货这一步的**产出**是「这批货给哪个店」（裁定 C4：计划不带店铺，排货时才定）。
★ 凡是某一步的产出，在后续每一步的表里都必须 NOT NULL。
```

**为什么是 NOT NULL 而不是运行时检查**：可空就等于**允许下游把它丢掉**，
而丢掉之后没有任何东西会报错 —— 下游只会少一笔货，而「少一笔」和「本来就没有」
在任何界面上都是同一个样子。`NOT NULL` 把它变成**写不进去**。

★ 这和 `line_id NOT NULL`（B4：没有计划记录的排货行写不进去）是同一条判据的两次应用。

#### 这条判据在本系统里的全部落点

| 步骤 | 它的产出 | 落在哪张表 | 约束 |
|---|---|---|---|
| 提交销售计划 | 计划采购量（货号级） | `plan_line.total_units` | NOT NULL |
| ★ **组采购单** | 「消耗的是哪个店那份需求」 | `po_source_map.seller_id` | ★ NOT NULL（§4.1） |
| ★ **排货** | 「这批货给哪个店」 | ★ `consignment_line.seller_id` | ★ **NOT NULL** |
| 排货 | 「从哪个仓出」 | `consignment_line.from_wid` | NOT NULL |
| ★ **投送 · 取得货件** | 亚马逊返回的货件号 | `ext_ref.doc_no`（`doc_kind='fba_shipment'`） | NOT NULL |
| ★ **投送 · 目的地明细** | 哪份货件去哪个 FC | ★ `consignment_split.ext_ref_id` | ★ **NOT NULL —— 在它自己那一层** |
| ★ **投送 · 回读归属** | 哪些货号进了哪个货件 | `consignment_split_line.units` | NOT NULL |

★★ **这条判据现在没有例外了**（O-2，2026-09-21）。

原先 `consignment_line.ext_shipment_no` 被列为「唯一的例外 · 可空」，理由是
「它是投送的产出，而排货行在投送之前就存在」。★ **那不是例外，是放错了地方** ——
把投送的产出塞进排货的对象里，于是不得不给它开一个可空的口子。

给它自己的表之后：货件号归 `ext_ref`（本来就有），**目的地明细归
`consignment_split` 且在那一层 NOT NULL**，排货行上则干脆没有这一列。★ **判据自洽了，而且「归属未知」变成了
「那张表里还没有行」——一个可以被查询、可以被点名的事实，不是一个 NULL。**

⚠️ 仍然成立的那半：把它当成「排货的产出」放进确认闸，就会得到一个
**没人填得出来的输入框**（负责人 2026-09-19 实测撞上，P15）——
根因是把 A-8「装箱数据人工填写」的适用范围错误地扩到了货件归属上。

#### 恒等式

```
★ Σ 排货网格上各格的已安排量  ≡  Σ 该排货表里所有池行与排货单行的量
```

★ 有了 `seller_id NOT NULL`，这条恒等式**结构上不可能被破坏** ——
每一笔货都必然落到网格的某一格上。
留着它作为断言，是为了挡住「网格的分组口径写错了」这另一类错。

## 6. ★ 量的视图与守恒约束

> ★ **所有量一律经视图读，不许各处自己写 SUM** ——
> 各处自己写，早晚有一处忘了过滤旁路终态（G4 那个洞）。

```sql
-- ★ 旁路终态（已撤销/已放弃）一律不参与求和，但映射永不删
CREATE VIEW scm.v_plan_line_qty AS
SELECT l.line_id, l.total_units,
  COALESCE((SELECT sum(m.units) FROM scm.po_source_map m
             JOIN scm.po p USING (po_id) WHERE m.line_id = l.line_id), 0)          AS confirmed_units,
  COALESCE((SELECT sum(m.units) FROM scm.po_source_map m
             JOIN scm.po p USING (po_id)
            WHERE m.line_id = l.line_id AND p.state = '已下单'), 0)                AS ordered_units,
  COALESCE((SELECT sum(d.units) FROM scm.dispatch_source_map d
             JOIN scm.consignment c USING (consignment_id)
            WHERE d.line_id = l.line_id AND c.our_stage IN ('草稿','已确认')), 0)   AS staged_units,
  COALESCE((SELECT sum(d.units) FROM scm.dispatch_source_map d
             JOIN scm.consignment c USING (consignment_id)
            WHERE d.line_id = l.line_id AND c.our_stage IN ('已投送','已归档')), 0) AS dispatched_units
FROM scm.plan_line l
WHERE l.state <> '已撤销';
```

### 6.1 约束触发器（★ 普通 CHECK 做不到跨行聚合）

```sql
CREATE FUNCTION scm.assert_line_qty_invariant() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE q record;
BEGIN
    SELECT * INTO q FROM scm.v_plan_line_qty
     WHERE line_id = COALESCE(NEW.line_id, OLD.line_id);
    IF q IS NULL THEN RETURN NULL; END IF;
    IF q.staged_units + q.dispatched_units > q.total_units THEN
        RAISE EXCEPTION '排货量超出需求：line_id=% 已排未发 %+已投送 % > 总量 %',
            q.line_id, q.staged_units, q.dispatched_units, q.total_units;
    END IF;
    IF q.ordered_units > q.confirmed_units OR q.confirmed_units > q.total_units THEN
        RAISE EXCEPTION '采购量违反 已下单 ≤ 已确认 ≤ 总量：line_id=%', q.line_id;
    END IF;
    RETURN NULL;
END $$;

CREATE CONSTRAINT TRIGGER c_dispatch_qty
    AFTER INSERT OR UPDATE OR DELETE ON scm.dispatch_source_map
    DEFERRABLE INITIALLY DEFERRED
    FOR EACH ROW EXECUTE FUNCTION scm.assert_line_qty_invariant();

CREATE CONSTRAINT TRIGGER c_po_qty
    AFTER INSERT OR UPDATE OR DELETE ON scm.po_source_map
    DEFERRABLE INITIALLY DEFERRED
    FOR EACH ROW EXECUTE FUNCTION scm.assert_line_qty_invariant();
```

★ **`DEFERRABLE INITIALLY DEFERRED` 是必须的**：
排货表「选择添加」会**重建整表**（先删后插），中间态必然短暂违反。
写成立即检查，正常操作会被自己的约束挡住。

### 6.2 ★ 补洞①：承重墙写入的前置条件（原本只在查询层）

> `04` §7.3 说「前置条件下沉库层」，但上一版只兑现了 `NOT NULL`。
> **候选查询是给人看的，约束是给系统守的** —— 直接调接口就能绕过候选。

```sql
CREATE FUNCTION scm.assert_po_source_precondition() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE st text; po_state text;
BEGIN
    IF TG_OP = 'DELETE' THEN
        SELECT p.state INTO po_state FROM scm.po p WHERE p.po_id = OLD.po_id;
        IF po_state <> '草稿' THEN
            RAISE EXCEPTION '采购单已下单，来源映射不可删（历史必须留着）';
        END IF;
        RETURN OLD;
    END IF;
    SELECT state INTO st FROM scm.plan_line WHERE line_id = NEW.line_id;
    IF st NOT IN ('已提交','已确认') THEN
        RAISE EXCEPTION '只有「已提交/已确认」的记录能组单，当前 %', st;
    END IF;
    RETURN NEW;
END $$;

CREATE FUNCTION scm.assert_dispatch_source_precondition() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE st text; sh_state text; cid bigint;
BEGIN
    cid := COALESCE(NEW.consignment_id, OLD.consignment_id);
    SELECT sh.state INTO sh_state
      FROM scm.consignment c JOIN scm.sheet sh USING (sheet_id)
     WHERE c.consignment_id = cid;
    IF sh_state <> '草稿' THEN
        RAISE EXCEPTION '排货表已确认，行不可增删改';
    END IF;
    IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
    SELECT state INTO st FROM scm.plan_line WHERE line_id = NEW.line_id;
    IF st NOT IN ('已下单','准备排货') THEN
        RAISE EXCEPTION '★ 只有「已下单/准备排货」的记录能排货，当前 % —— 这批货可能还不存在', st;
    END IF;
    RETURN NEW;
END $$;

CREATE TRIGGER t_po_src_pre     BEFORE INSERT OR UPDATE OR DELETE ON scm.po_source_map
    FOR EACH ROW EXECUTE FUNCTION scm.assert_po_source_precondition();
CREATE TRIGGER t_disp_src_pre   BEFORE INSERT OR UPDATE OR DELETE ON scm.dispatch_source_map
    FOR EACH ROW EXECUTE FUNCTION scm.assert_dispatch_source_precondition();
```

★ **`dispatch` 那条是 `04` §7.0 里 H3 点名「最危险」的那个洞**：
不挡住，「还没下单的货」就能被排出去 —— 而那批货**根本不存在**。
确认闸 G3 只校验本地仓可发量，若仓里恰好有别批次的同款货，**G3 会放行，于是别人的货被发走了**。

### 6.3 ★ 补洞②：领星进度的单调性 —— 不用守卫，改成派生

> 上一版把「只许前进」放在应用层（回执器），
> 而 `04` §7.7 自己写着「应用层的约束只对记得调它的那条路径有效」—— **自相矛盾**。
>
> ★ **写 DDL 时发现根本不需要守卫：把当前阶段改成派生就够了。**

```sql
CREATE VIEW scm.v_ext_current_stage AS
SELECT ext_ref_id,
       CASE
         -- ① 旁路终态一旦观察到就是终局，不参与 rank 比较
         WHEN bool_or(stage IN ('已撤销','已驳回')) THEN
              (array_agg(stage ORDER BY observed_on DESC)
               FILTER (WHERE stage IN ('已撤销','已驳回')))[1]
         -- ② 其余取 rank 最大的那条，★ 不是最新的那条
         ELSE (array_agg(stage ORDER BY stage_rank DESC, observed_on DESC))[1]
       END AS stage,
       max(stage_rank)        AS stage_rank,
       max(observed_on)       AS last_observed_on
FROM scm.ext_stage_observation
GROUP BY ext_ref_id;
```

| 原方案 | 现方案 |
|---|---|
| 存物化列 + 应用层守卫「倒退不写入」 | ★ **不存物化列，当前阶段 = `max(stage_rank)` 派生** |
| 倒退的观察被**丢掉** | 倒退的观察**照常写入，但不影响当前阶段** |
| 守卫漏调 = 状态真的倒退 | ★ **结构上不可能倒退** —— 没有守卫可漏 |

★ **而且证据留住了**：CH 掉档那天的观察还在表里，
下次再掉档能查出规律。原方案「倒退就不写入」把证据也一起丢了。

### 6.4 ★ E-2：格子与其 msku 构成的守恒

```sql
CREATE FUNCTION scm.assert_cell_msku_sum() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE cell_units int; sum_units int;
BEGIN
    SELECT c.system_units,
           COALESCE((SELECT sum(m.system_units) FROM scm.plan_cell_msku m
                      WHERE (m.plan_id,m.seller_id,m.sku,m.period_start)
                          = (c.plan_id,c.seller_id,c.sku,c.period_start)), 0)
      INTO cell_units, sum_units
      FROM scm.plan_cell c
     WHERE (c.plan_id,c.seller_id,c.sku,c.period_start)
         = (COALESCE(NEW.plan_id,OLD.plan_id), COALESCE(NEW.seller_id,OLD.seller_id),
            COALESCE(NEW.sku,OLD.sku),         COALESCE(NEW.period_start,OLD.period_start));
    IF cell_units IS NOT NULL AND cell_units <> sum_units THEN
        RAISE EXCEPTION '格子预估 % 与其 msku 构成之和 % 不符 —— 构成不是第二个真相',
            cell_units, sum_units;
    END IF;
    RETURN NULL;
END $$;

    AFTER INSERT OR UPDATE OR DELETE ON scm.plan_cell_msku
```

## 7. ⑥ 预测域 —— ★ 负责人 Excel 的逐格落库

> **这一节存在的唯一理由**
>
> 负责人原话：**「数据库里必须要有我这张 Excel 的数据字段统计记录，
> 因为我必须要能从数据库里统计该表！否则我根本无法核实你计算的准确性！」**
>
> ★ 所以判据不是「算得对」，是 **「算完之后，别人能不能自己查出来核对」**。
> 只有结论没有构成的预测，**没人能证伪** —— 而没人能证伪的数，
> 迟早会被当成事实引用。
>
> 模型本身见 `14-库存预测数学模型.md`。这一节只管**怎么存、怎么查、怎么核**。

### 7.0 ★ 最要紧的一条：没有「合计表」

`14` §1.1 恒等式① 是 **`可售合计 ≡ Σ_店铺 可售`**。

★ **这条不靠断言保证，靠「根本不存在合计表」保证** ——
合计只由视图 `v_forecast_sheet` 现算。存了合计就是第二个真相，
而两个真相迟早会分叉，分叉之后没人知道该信哪个。

| 落库的 | 不落库的 |
|---|---|
| 店铺×msku×月（可售侧最细颗粒度） | ★ 货号合计 —— 视图现算 |
| 货号×段×月（管道，无店铺的那两段） | ★ 「在途合计」—— 视图现算 |

#### 36 `forecast_cell` —— 可售侧最细颗粒度

```sql
CREATE TABLE scm.forecast_cell (
    run_id             bigint  NOT NULL REFERENCES scm.forecast_run(run_id) ON DELETE CASCADE,
    seller_id          text    NOT NULL REFERENCES scm.seller(seller_id),
    seller_sku         text    NOT NULL,
    sku                text    NOT NULL REFERENCES scm.sku_catalog(sku),
    period             date    NOT NULL,              -- 月首日
    opening            integer NOT NULL,
    system_units       integer,                       -- 系统销量预估（建议值）
    -- ★ 这个预估是**外推**出来的（超出预估覆盖范围，沿用最后一个值）。
    --   当 NULL 处理会让推演从那月起一律未知、整条曲线作废；
    --   不标记又等于把外推冒充成预估。见 `14` §5。
    system_extrapolated boolean NOT NULL DEFAULT false,
    demand_expected    integer,                       -- ★ NULL = 未知，**不是 0**
    inbound_confirmed  integer NOT NULL DEFAULT 0,    -- ★ 事实：已发运，有预计到货日
    inbound_planned    integer NOT NULL DEFAULT 0,    -- ★ 假设：由计划采购推算
    closing            integer,                       -- NULL ⟺ unknown
    unknown            boolean NOT NULL DEFAULT false,
    PRIMARY KEY (run_id, seller_id, seller_sku, period),
    -- ★ 没有预估就谈不上外推 —— 防止这个标记变成一个谁都没在维护的布尔
    CONSTRAINT ck_cell_extrapolated CHECK (NOT system_extrapolated OR system_units IS NOT NULL),

    -- ★ 恒等式③ 直接做成 CHECK：这一层不成立，就是漏了一个流向
    CONSTRAINT ck_cell_identity CHECK (
        (unknown AND closing IS NULL)
        OR closing = opening - COALESCE(demand_expected, 0)
                   + inbound_confirmed + inbound_planned
    ),
    CONSTRAINT ck_cell_nonneg CHECK (inbound_confirmed >= 0 AND inbound_planned >= 0)
);
```

★ **`closing` 不做 `>= 0` 约束** —— 负库存就是「缺多少」，
截断成 0 会把整份预测里最重要的那个数抹掉（`14` B8）。

#### 37 `forecast_pipeline` —— 管道四段

```sql
CREATE TABLE scm.forecast_pipeline (
    run_id     bigint  NOT NULL REFERENCES scm.forecast_run(run_id) ON DELETE CASCADE,
    sku        text    NOT NULL REFERENCES scm.sku_catalog(sku),
    segment    text    NOT NULL REFERENCES scm.pipeline_segment(segment),
    seller_id  text        NULL REFERENCES scm.seller(seller_id),
    period     date    NOT NULL,
    opening    integer NOT NULL,
    inflow     integer NOT NULL,
    outflow    integer NOT NULL,
    closing    integer NOT NULL,
    seller_key text GENERATED ALWAYS AS (COALESCE(seller_id, '')) STORED,
    PRIMARY KEY (run_id, sku, segment, seller_key, period),

    -- ★ 「排货这条线以下没有店铺」—— 从注释变成约束
    CONSTRAINT ck_pipe_directed CHECK (
        (segment IN ('making', 'local')     AND seller_id IS NULL)
     OR (segment IN ('staged', 'shipping')  AND seller_id IS NOT NULL)
    ),
    CONSTRAINT ck_pipe_identity CHECK (closing = opening + inflow - outflow),
    CONSTRAINT ck_pipe_nonneg   CHECK (opening >= 0 AND inflow >= 0
                                   AND outflow >= 0 AND closing >= 0)
);
```

★ 管道量**必须非负** —— 与可售库存相反。可售可以欠着（缺货），
但「生产中 −5 件」没有任何物理含义，出现就是算错了。

> ⚠️ `ck_pipe_directed` 把段名写死在 CHECK 里，与 `pipeline_segment.directed` 重复。
> 加段时**两处都要改**。做成触发器引用白名单更干净，但普通 CHECK 快得多 ——
> 段只有四个且不常变，这里选了快的。**加段时漏改一处会静默放过错数据，已记入 §9。**

#### ★ 天 → 月：锚在月中

计划只说「10 月下单」，没说几号。落回月份要选锚点，**取该月 15 号**：

```sql
-- 从某月起算 n 天，落在哪个月
CREATE FUNCTION scm.month_after_days(period date, days int)
RETURNS date LANGUAGE sql IMMUTABLE AS $$
    SELECT date_trunc('month',
        date_trunc('month', period) + INTERVAL '14 days' + (days || ' days')::interval)::date
$$;
```

★ 真实下单日在当月大致均匀分布，**期望值就是月中** —— 而这整套本来就是期望值模型。
锚月初系统性乐观（把每笔单都当 1 号下的），锚月末又会把「国内调拨 3 天」
一律推到下个月，**抹掉国内最该看见的「当月就到」**。理由详见 `14` §2.3。

```sql
CREATE VIEW scm.v_forecast_sheet AS
WITH agg AS (                                     -- ★ 合计由此现算，不落表
    SELECT run_id, sku, period,
           SUM(opening)                     AS opening,
           SUM(demand_expected)             AS demand_expected,
           bool_or(demand_expected IS NULL) AS demand_has_unfilled,   -- ★ 见下
           SUM(inbound_confirmed)           AS inbound_confirmed,
           SUM(inbound_planned)             AS inbound_planned,
           SUM(closing)                     AS closing,
           bool_or(unknown)                 AS unknown
      FROM scm.forecast_cell
     GROUP BY run_id, sku, period
), pipe AS (
    SELECT run_id, sku, period,
           SUM(closing) FILTER (WHERE segment = 'making')   AS making,
           SUM(closing) FILTER (WHERE segment = 'local')    AS local_stock,
           SUM(closing) FILTER (WHERE segment = 'staged')   AS staged,
           SUM(closing) FILTER (WHERE segment = 'shipping') AS shipping,
           SUM(closing)                                     AS in_pipeline_total,
           SUM(inflow)  FILTER (WHERE segment = 'making')   AS purchase_planned
      FROM scm.forecast_pipeline
     GROUP BY run_id, sku, period
)
SELECT a.run_id, a.sku, a.period,
       a.opening                                   AS "期初库存",
       a.demand_expected                           AS "期望销量",
       a.demand_has_unfilled                       AS "期望销量有未填",
       a.inbound_confirmed                         AS "当期入库·已确认",
       a.inbound_planned                           AS "当期入库·按计划推算",
       a.inbound_confirmed + a.inbound_planned     AS "当期入库量",
       a.closing                                   AS "期末库存",
       a.unknown                                   AS "含未知",
       p.purchase_planned                          AS "计划采购量",
       p.making                                    AS "生产中",
       p.local_stock                               AS "本地仓在库",
       p.staged                                    AS "排货中",
       p.shipping                                  AS "货运中",
       p.in_pipeline_total                         AS "在途合计"
  FROM agg a LEFT JOIN pipe p USING (run_id, sku, period);
```

★ **`demand_has_unfilled` 这一列不是装饰。**
`SUM()` 遇 NULL 直接跳过 —— 不把它标出来，**「有一个店没填」就会被静默读成
「合计少了一点」**。这正是本项目踩过六次的「静默丢失」：
**检查任何过滤/聚合，必须同时统计被丢掉的那一侧。**

### 7.3 ★ 核对视图：返回 0 行才算对

```sql
CREATE VIEW scm.v_forecast_check AS
-- ① 恒等式④：期初[m] ≡ 期末[m−1]（跨行，CHECK 做不到）
SELECT 'cell_rollover' AS rule, c.run_id, c.sku,
       c.seller_id || '/' || c.seller_sku AS grain, c.period,
       c.opening AS got, prev.closing AS want
  FROM scm.forecast_cell c
  JOIN scm.forecast_cell prev
    ON prev.run_id = c.run_id AND prev.seller_id = c.seller_id
   AND prev.seller_sku = c.seller_sku
   AND prev.period = c.period - interval '1 month'
 WHERE c.opening IS DISTINCT FROM prev.closing

UNION ALL
-- ② 管道同上
SELECT 'pipe_rollover', p.run_id, p.sku, p.segment || '/' || p.seller_key, p.period,
       p.opening, prev.closing
  FROM scm.forecast_pipeline p
  JOIN scm.forecast_pipeline prev
    ON prev.run_id = p.run_id AND prev.sku = p.sku AND prev.segment = p.segment
   AND prev.seller_key = p.seller_key
   AND prev.period = p.period - interval '1 month'
 WHERE p.opening IS DISTINCT FROM prev.closing

UNION ALL
-- ③ ★ `14` B9：某段的流出 ≡ 下一段的流入。对不上就是漏了货
SELECT 'segment_chain', cur.run_id, cur.sku, cur.segment || '→' || nxt.segment, cur.period,
       cur.outflow, nxt.inflow
  FROM scm.forecast_pipeline cur
  JOIN scm.pipeline_segment  sc  ON sc.segment  = cur.segment
  JOIN scm.pipeline_segment  sn  ON sn.ord      = sc.ord + 1
  JOIN LATERAL (
        SELECT COALESCE(SUM(inflow), 0) AS inflow
          FROM scm.forecast_pipeline n
         WHERE n.run_id = cur.run_id AND n.sku = cur.sku
           AND n.segment = sn.segment AND n.period = cur.period
       ) nxt ON true
 WHERE cur.outflow IS DISTINCT FROM nxt.inflow

UNION ALL
-- ④ ★ 末段流出 ≡ 各店铺当期入库（管道与可售的接缝）
SELECT 'pipe_to_cell', p.run_id, p.sku, 'shipping→cell', p.period,
       SUM(p.outflow), c.inbound
  FROM scm.forecast_pipeline p
  JOIN LATERAL (
        SELECT COALESCE(SUM(inbound_confirmed + inbound_planned), 0) AS inbound
          FROM scm.forecast_cell fc
         WHERE fc.run_id = p.run_id AND fc.sku = p.sku AND fc.period = p.period
       ) c ON true
 WHERE p.segment = 'shipping'
 GROUP BY p.run_id, p.sku, p.period, c.inbound
HAVING SUM(p.outflow) IS DISTINCT FROM c.inbound

UNION ALL
-- ⑤ ★ 推算入库 ≡ 其构成明细之和（有结论必有依据）
SELECT 'inbound_trace', c.run_id, c.sku,
       c.seller_id || '/' || c.seller_sku, c.period,
       c.inbound_planned, COALESCE(t.units, 0)
  FROM scm.forecast_cell c
  LEFT JOIN LATERAL (
        SELECT SUM(units) AS units FROM scm.forecast_inbound_trace x
         WHERE x.run_id = c.run_id AND x.seller_id = c.seller_id
           AND x.seller_sku = c.seller_sku AND x.arrive_period = c.period
           AND x.source = 'allocated'
       ) t ON true
 WHERE c.inbound_planned IS DISTINCT FROM COALESCE(t.units, 0);
```

★ **每一轮推演结束必须跑它，非空就整轮作废** ——
不作废的话，一个能自检却没自检的系统，比没有自检更糟：
它给了人「已经查过了」的错觉。
